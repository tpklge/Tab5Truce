cmake_minimum_required(VERSION 3.20)
project(truce_ui_simulator C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(LVGL_DIR "${ROOT}/.pio/libdeps/m5stack-tab5/lvgl" CACHE PATH "LVGL 9.3 source")
set(LV_BUILD_CONF_DIR "${ROOT}/include" CACHE PATH "" FORCE)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_compile_definitions(TRUCE_HOST=1 LV_CONF_INCLUDE_SIMPLE=1
  LV_USE_FS_STDIO=1 LV_FS_STDIO_LETTER=84 LV_FS_STDIO_PATH="${ROOT}/sdcard/truce")
include_directories("${ROOT}/include")
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
add_link_options(-fsanitize=address,undefined)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(truce_simulator simulator.cpp "${ROOT}/src/core/truce/ui_view.cpp")
target_include_directories(truce_simulator PRIVATE "${ROOT}/src/core/truce")
target_link_libraries(truce_simulator PRIVATE lvgl)
enable_testing()
add_test(NAME ui_and_storage COMMAND truce_simulator "${CMAKE_BINARY_DIR}/screenshots")
add_executable(truce_radio_transport radio_transport.cpp "${ROOT}/src/core/truce/radio_transport.cpp")
target_include_directories(truce_radio_transport BEFORE PRIVATE stubs)
target_compile_definitions(truce_radio_transport PRIVATE TRUCE_LVGL=1)
add_test(NAME radio_transport_failure_and_retry COMMAND truce_radio_transport)
